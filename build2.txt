              private fun proj(lat: Float, lon: Float, cx: Float, cy: Float, r: Float): Boolean {
                  val la = Math.toRadians(lat.toDouble())
                  val lo = Math.toRadians((lon + rot).toDouble())
                  val x = cos(la) * sin(lo)
                  val y0 = sin(la)
                  val z0 = cos(la) * cos(lo)
                  val a = 0.4
                  val y = y0 * cos(a) - z0 * sin(a)
                  val z = y0 * sin(a) + z0 * cos(a)
                  var px = x.toFloat()
                  var py = y.toFloat()
                  if (z < 0) {
                      val l = hypot(px, py)
                      if (l > 0.0001f) { px /= l; py /= l } else { px = 1f; py = 0f }
                  }
                  pt[0] = cx + r * px
                  pt[1] = cy - r * py
                  return z >= 0
              }

              override fun onDraw(cv: Canvas) {
                  val now = System.nanoTime()
                  val dt = if (last == 0L) 0.016f else min(0.05f, (now - last) / 1e9f)

                  val st = state
                  rot = (rot + (if (st == 1) 60f else if (st == 2) 40f else 14f) * dt) % 360f

                  cv.drawCircle(cx, cy, r * 1.9f, p)
                  p.shader = null

                  if (st == 1) {
                      p.style = Paint.Style.STROKE
                      p.strokeWidth = 4f
                      for (i in 0..1) {
                          val ph = (t * 0.8f + i * 0.5f) % 1f
                          p.color = Color.argb(((1f - ph) * 130).toInt(), 140, 190, 255)
                          cv.drawCircle(cx, cy, r * (1.05f + 0.5f * ph), p)
                      }
                      p.style = Paint.Style.FILL
                  }

                  val clip = Path()
                  clip.addCircle(cx, cy, r, Path.Direction.CW)
                  cv.save()
                  cv.clipPath(clip)
                  p.shader = RadialGradient(cx - r * 0.35f, cy - r * 0.4f, r * 1.6f, intArrayOf(Color.rgb(127, 200, 255), Color.rgb(36, 98, 214), Color.rgb(8, 30, 100)), floatArrayOf(0f, 0.55f, 1f), Shader.TileMode.CLAMP)
                  cv.drawCircle(cx, cy, r, p)
                  p.shader = null
                  p.color = Color.rgb(63, 174, 90)
                  for (poly in land) {
                      val path = Path()
                      var any = false
                      var i = 0
                      while (i < poly.size) {
                          if (proj(poly[i], poly[i + 1], cx, cy, r)) any = true
                          if (i == 0) path.moveTo(pt[0], pt[1]) else path.lineTo(pt[0], pt[1])
                          i += 2
                      }
                      path.close()
                      if (any) cv.drawPath(path, p)
                  }
                  p.shader = RadialGradient(cx, cy, r, intArrayOf(Color.argb(0, 0, 0, 0), Color.argb(0, 0, 0, 0), Color.argb(150, 0, 10, 40)), floatArrayOf(0f, 0.65f, 1f), Shader.TileMode.CLAMP)

                  cv.restore()
                  postInvalidateOnAnimation()
              }

              private fun findContact(name: String): Pair<String, String>? {
                  if (checkSelfPermission(Manifest.permission.READ_CONTACTS) != PackageManager.PERMISSION_GRANTED) return null
                  val n = clean(name)
                  if (n.isEmpty()) return null
                  val c = contentResolver.query(ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
                      arrayOf(ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME, ContactsContract.CommonDataKinds.Phone.NUMBER), null, null, null) ?: return null
                  c.use {
                      while (it.moveToNext()) {
                          val dn = it.getString(0) ?: continue
                          val low = dn.lowercase()
                          if (low.contains(n) || low.replace(" ", "") == n.replace(" ", "")) return Pair(dn, it.getString(1) ?: "")
                      }
                  }
                  return null
              }

                  c.setRequestProperty("Content-Type", "application/json")
                  val body = JSONObject()
                      .put("system_instruction", JSONObject().put("parts", JSONArray().put(JSONObject().put("text", PERSONA))))
                      .put("contents", JSONArray().put(JSONObject().put("role", "user").put("parts", JSONArray().put(JSONObject().put("text", q)))))
                  c.outputStream.use { it.write(body.toString().toByteArray()) }
                  JSONObject(c.inputStream.bufferedReader().readText()).getJSONArray("candidates").getJSONObject(0)
                      .getJSONObject("content").getJSONArray("parts").getJSONObject(0).getString("text").trim()

              private fun handle(raw: String) {
                  val q = raw.lowercase().trim().replace("chat gpt", "chatgpt")
                  log("Tum", raw)
                  fun has(vararg w: String) = w.any { q.contains(it) }

                  // chat kholo / chat me message likho
                  Regex("^(?:open\\s+|kholo\\s+)?(.+?)\\s+(?:ki\\s+|ka\\s+)?chat\\b(.*)$").find(q)?.let { m ->
                      val rest = m.groupValues[2]
                      val mm = Regex("(?:likho|write|type|bhejo|send|bolo|kaho)\\s+(.+)").find(rest)
                          ?: Regex("(?:me|mein|par|pe)?\\s*(.+?)\\s+(?:likho|write|type|bhejo|send)\\s*$").find(rest)
                      val msg = mm?.groupValues?.get(1)?.replace(Regex("^(ki|ke|that)\\s+"), "")?.trim() ?: ""
                      val c = findContact(m.groupValues[1])
                      if (c == null) { say("Mujhe ${clean(m.groupValues[1])} naam ka contact nahi mila"); return }
                      whatsapp(c, msg); return
                  }
                  // X ko message bhejo
                  Regex("(?:whatsapp\\s*(?:pe|par)?\\s*)?(.+?)\\s+ko\\s+(?:whatsapp\\s*(?:pe|par)?\\s*)?(?:message|msg|sms)?\\s*(?:bhejo|bhej do|bhej|send|likho|bolo|kaho|type)\\s*(?:ki|ke)?\\s*(.+)").find(q)?.let { m ->
                      val c = findContact(m.groupValues[1])
                      if (c == null) { say("Mujhe ${clean(m.groupValues[1])} naam ka contact nahi mila"); return }
                      val msg = m.groupValues[2].trim()
                      if (has("sms")) {
                          if (checkSelfPermission(Manifest.permission.SEND_SMS) == PackageManager.PERMISSION_GRANTED) {
                              @Suppress("DEPRECATION") SmsManager.getDefault().sendTextMessage(c.second, null, msg, null, null)
                              say("${c.first} ko SMS bhej diya")
                          } else say("SMS ki permission allow karo")
                      } else whatsapp(c, msg)
                      return
                  }
                  // call
                  Regex("(.+?)\\s+ko\\s+(?:call|phone)").find(q)?.let { m ->
                      val c = findContact(m.groupValues[1])
                      if (c == null) { say("Ye contact mujhe nahi mila"); return }
                      say("${c.first} ko call laga rahi hu")
                      val act = if (checkSelfPermission(Manifest.permission.CALL_PHONE) == PackageManager.PERMISSION_GRANTED) Intent.ACTION_CALL else Intent.ACTION_DIAL
                      go(Intent(act, Uri.parse("tel:" + c.second))); return
                  }
                  // alarm / timer
                  val al = Regex("(\\d{1,2})(?::(\\d{2}))?\\s*(?:baje|bje|am|pm)").find(q)
                  if (al != null && has("alarm")) {
                      var h = al.groupValues[1].toInt()
                      val mi = al.groupValues[2].ifEmpty { "0" }.toInt()
                      if (has("pm", "shaam", "raat", "dopahar") && h < 12) h += 12
                      go(Intent(AlarmClock.ACTION_SET_ALARM).putExtra(AlarmClock.EXTRA_HOUR, h).putExtra(AlarmClock.EXTRA_MINUTES, mi).putExtra(AlarmClock.EXTRA_SKIP_UI, true))
                      say("Alarm $h baj ke $mi minute par set kar diya"); return
                  }
                  val tm = Regex("(\\d+)\\s*(second|sec|minute|min|mint|ghante|ghanta|hour)").find(q)
                  if (tm != null && has("timer")) {
                      val n = tm.groupValues[1].toInt(); val u = tm.groupValues[2]
                      val secs = n * (if (u.startsWith("sec")) 1 else if (u.startsWith("min")) 60 else 3600)
                      go(Intent(AlarmClock.ACTION_SET_TIMER).putExtra(AlarmClock.EXTRA_LENGTH, secs).putExtra(AlarmClock.EXTRA_SKIP_UI, true))
                      say("$n ${tm.groupValues[2]} ka timer laga diya"); return
                  }
                  // phone controls
                  if (has("torch", "flashlight")) {
                      val on = !has("band", "off", "bujha")
                      try { val cm = getSystemService(Context.CAMERA_SERVICE) as CameraManager; cm.setTorchMode(cm.cameraIdList[0], on); say(if (on) "Torch chalu kar di" else "Torch band kar di") }
                      catch (e: Exception) { say("Torch nahi chal paayi") }
                      return
                  }
                  if (has("wifi", "wi-fi")) { say("WiFi settings khol rahi hu"); go(Intent(Settings.ACTION_WIFI_SETTINGS)); return }
                  if (has("bluetooth")) { say("Bluetooth settings khol rahi hu"); go(Intent(Settings.ACTION_BLUETOOTH_SETTINGS)); return }
                  if (has("camera", "selfie")) { say("Camera khol rahi hu"); go(Intent(MediaStore.ACTION_IMAGE_CAPTURE)); return }
                  if (has("settings")) { say("Settings khol rahi hu"); go(Intent(Settings.ACTION_SETTINGS)); return }
                  // open any app
                  if (has("open", "kholo", "khol", "chalu", "start", "launch") && !has("chala do", "bajao", "gana", "song", "chalao")) {
                      val nm = q.replace(Regex("\\b(open|kholo|khol|chalu|start|launch|app|noah|please|karo|kar|do|de|ko|ek|ye)\\b"), " ").replace(Regex("\\s+"), " ").trim()
                      if (nm.isNotEmpty()) { openApp(nm); return }
                  }
                  // YouTube play
                  if (has("youtube", "chala", "bajao", "gana", "gaana", "play ")) {
                      val term = q.replace(Regex("\\b(noah|please|youtube|you tube|pe|par|on|me|mein|ko|ka|ki|ke|chala|chalao|bajao|baja|play|do|de|karo|kar|kro|search|video|dikhao)\\b"), " ").replace(Regex("\\s+"), " ").trim()
                      if (term.isEmpty()) { openApp("youtube"); return }
                      say("$term chala rahi hu")
                      thread {
                          val id = ytFind(term)
                          runOnUiThread {
                              val url = if (id != null) "https://www.youtube.com/watch?v=$id" else "https://www.youtube.com/results?search_query=" + enc(term)
                              try { startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(url)).setPackage("com.google.android.youtube")) }
                              catch (e: Exception) { go(Intent(Intent.ACTION_VIEW, Uri.parse(url))) }
                          }
                      }
                      return
                  }
                  if (has("time", "samay", "baje kya")) { say("Abhi " + SimpleDateFormat("hh:mm a", Locale.getDefault()).format(Date()) + " ho rahe hain"); return }
                  if (has("i love you", "love you", "pyar", "pyaar")) { say("Main bhi tumse bahut pyaar karti hu"); return }
                  if (has("kaisi ho", "kaise ho", "how are you")) { say("Main bahut achhi hu, tumse baat jo ho rahi hai. Tum batao?"); return }
                  if (has("naam", "who are you", "kaun ho")) { say("Main Noah hu, tumhari apni dost aur assistant"); return }
                  if (has("thank", "shukriya")) { say("Arre, thank you kaisa. Tumhare liye toh kuch bhi"); return }
                  if (has("hello", "hi ", "hey", "namaste") && q.length < 12) { say("Hello jaan! Bolo, kya karu tumhare liye?"); return }
                  // AI fallback
                  val key = getSharedPreferences("noah", MODE_PRIVATE).getString("key", "") ?: ""
                  if (key.isBlank()) { say("Samajh nahi aaya. Smart jawab ke liye AI key daalo."); return }
                  thread { val a = gemini(raw, key); runOnUiThread { say(a ?: "Abhi jawab nahi mil paya") } }
              }
          }

          package com.noah.app

          import android.accessibilityservice.AccessibilityService
          import android.content.Intent
          import android.view.accessibility.AccessibilityEvent
          import android.view.accessibility.AccessibilityNodeInfo

          class NoahService : AccessibilityService() {
              companion object {
                  @Volatile var instance: NoahService? = null
                  @Volatile var pendingUntil = 0L
              }

              override fun onServiceConnected() { instance = this }

              override fun onUnbind(intent: Intent?): Boolean {
                  instance = null
                  return super.onUnbind(intent)
              }

              override fun onInterrupt() {}

              override fun onAccessibilityEvent(e: AccessibilityEvent?) {
                  if (System.currentTimeMillis() > pendingUntil) return
                  val root = rootInActiveWindow ?: return
                  val list = root.findAccessibilityNodeInfosByViewId("com.whatsapp:id/send")
                  var target: AccessibilityNodeInfo? = list?.firstOrNull()
                  if (target == null) target = findByDesc(root, "send")
                  if (target != null && target.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
                      pendingUntil = 0L
                  }
              }

              private fun findByDesc(n: AccessibilityNodeInfo?, d: String): AccessibilityNodeInfo? {
                  if (n == null) return null
                  val cd = n.contentDescription?.toString()?.lowercase()
                  if (cd == d && n.isClickable) return n
                  for (i in 0 until n.childCount) {
                      val r = findByDesc(n.getChild(i), d)
                      if (r != null) return r
                  }
                  return null
              }
          }
